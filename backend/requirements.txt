# ⚠️  TORCH IS INSTALLED SEPARATELY — see backend/Dockerfile.
# The Dockerfile installs the CPU-only wheel from PyTorch's CPU index BEFORE
# this file is processed. Without that, pip resolves the default CUDA build and
# downloads 3-5 GB of NVIDIA libraries we never use.
#
# Running OUTSIDE Docker? Install torch first, the same way:
#   pip install torch==2.4.1 --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements.txt
torch==2.4.1

# These are transitive deps of sentence-transformers that it does NOT pin.
# Left unpinned, pip resolves whatever is newest — which is how this build
# ended up on torch 2.13 + transformers 4.57 + numpy 2.5, a combination nobody
# has tested together. Pinning them makes builds reproducible.

numpy==1.26.4
