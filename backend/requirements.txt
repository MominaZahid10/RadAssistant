# ── Database (PostgreSQL) ─────────────────────────────────────
sqlalchemy[asyncio]==2.0.30  # ORM — write Python instead of raw SQL

# ── Configuration ─────────────────────────────────────────────
pydantic-settings==2.3.0  # Read settings from .env files with validation
python-dotenv==1.0.1      # Load .env files into environment variables

# ── Vector Database ───────────────────────────────────────────
qdrant-client==1.10.0     # Client to talk to Qdrant vector database

# ── LLM Providers (Phase 3) ───────────────────────────────
groq==0.9.0               # Groq API client — free tier, fast inference
mistralai==1.2.5           # Mistral AI client (v2 SDK) — quality models
openai==1.40.0            # OpenAI client — production provider

# ── Document Processing (Phase 2) ────────────────────────────
pymupdf==1.24.0           # PDF text extraction — fastest Python PDF lib (C-based MuPDF engine)
python-docx==1.1.0        # DOCX (Word) text extraction
Pillow==10.4.0            # Image handling — open, resize, convert image formats
pytesseract==0.3.13       # OCR — extract text from images (uses Tesseract engine)

# ── Medical Imaging (Phase 4) ────────────────────────────────
# DICOM parsing. Imported lazily by dicom_service — if it's absent the
# app still runs and DICOM uploads are rejected with a clear message,
# so a failed install never takes the service down.
pydicom==2.4.4
# Decoders for compressed transfer syntaxes (JPEG-LS, JPEG2000), which
# most real CT/MR exports use. Without these, pixel_array raises.
pylibjpeg==2.0.0
pylibjpeg-libjpeg==2.1.0
pylibjpeg-openjpeg==2.1.1

# ── Embeddings (Phase 2) ─────────────────────────────────────
sentence-transformers==3.0.1  # Load & run embedding models (all-MiniLM-L6-v2)

# ⚠️  TORCH IS INSTALLED SEPARATELY — see backend/Dockerfile.
# The Dockerfile installs the CPU-only wheel from PyTorch's CPU index BEFORE
# this file is processed. Without that, pip resolves the default CUDA build and
# downloads 3-5 GB of NVIDIA libraries we never use.
#
# Running OUTSIDE Docker? Install torch first, the same way:
#   pip install torch==2.4.1 --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements.txt
torch==2.4.1

# These are transitive deps of sentence-transformers that it does NOT pin.
# Left unpinned, pip resolves whatever is newest — which is how this build
# ended up on torch 2.13 + transformers 4.57 + numpy 2.5, a combination nobody
# has tested together. Pinning them makes builds reproducible.
transformers==4.44.2
numpy==1.26.4

# ── File Upload Support ──────────────────────────────────────
python-multipart==0.0.9   # Required by FastAPI to accept file uploads (multipart/form-data)
aiofiles==24.1.0          # Async file I/O — needed for saving uploaded files efficiently

# ── Medical Knowledge Sources (Phase 2) ──────────────────────
biopython==1.84           # Access NCBI/PubMed APIs (E-utilities) — fetch StatPearls articles
httpx==0.27.0             # Modern async HTTP client — for fetching external medical content

# ── Testing ──────────────────────────────────────────────────
# Run with: cd backend && pytest tests/ -v
pytest==8.3.2             # Test runner
pytest-asyncio==0.24.0    # Lets pytest run `async def` tests (needed from Phase 3)
